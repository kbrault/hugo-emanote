{{- /* Internal links inherit the note-title strip's palette at one font-weight step lighter, reading as small inline "chips" that name the linked page — the signature look of the source theme's prose. External links keep plain-text color with an underline and a small glyph on hover; mailto gets a distinct mail glyph. */ -}} {{- $dest := .Destination -}} {{- $isMail := strings.HasPrefix $dest "mailto:" -}} {{- $isAbs := or (strings.HasPrefix $dest "http://") (strings.HasPrefix $dest "https://") -}} {{- $isExternal := false -}} {{- if and $isAbs (not $isMail) -}} {{- if not (strings.HasPrefix $dest site.BaseURL) -}} {{- $isExternal = true -}} {{- end -}} {{- end -}} {{- if $isMail -}} {{ .Text | safeHTML }} {{- else if $isExternal -}} {{ .Text | safeHTML }} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
